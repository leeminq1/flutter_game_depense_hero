```dart
import 'dart:ui';

class BattlefieldCameraSnapshot {
  const BattlefieldCameraSnapshot({
    this.zoom = 1,
    this.pan = Offset.zero,
  });

  final double zoom;
  final Offset pan;

  bool get isTransformed => (zoom - 1).abs() > 0.001 || pan.distance > 0.5;

  @override
  bool operator ==(Object other) =>
      other is BattlefieldCameraSnapshot && other.zoom == zoom && other.pan == pan;

  @override
  int get hashCode => Object.hash(zoom, pan);
}

class BattlefieldCameraTransform {
  static const minZoom = 0.7;
  static const maxZoom = 2.5;
  static const dragThreshold = 8.0;

  double _zoom = 1;
  Offset _pan = Offset.zero;
  Offset _gestureStart = Offset.zero;
  Offset _lastFocalPoint = Offset.zero;
  bool _dragging = false;
  bool _pinching = false;

  double get zoom => _zoom;
  Offset get pan => _pan;
  bool get suppressTap => _dragging || _pinching;
  bool get isTransformed => snapshot.isTransformed;
  BattlefieldCameraSnapshot get snapshot => BattlefieldCameraSnapshot(zoom: _zoom, pan: _pan);

  void beginGesture(Offset focalPoint) {
    _gestureStart = focalPoint;
    _lastFocalPoint = focalPoint;
    _dragging = false;
    _pinching = false;
  }

  bool updatePan(Offset focalPoint) {
    if (!_dragging && (focalPoint - _gestureStart).distance <= dragThreshold) {
      _lastFocalPoint = focalPoint;
      return false;
    }
    _dragging = true;
    _pan += (focalPoint - _lastFocalPoint) / _zoom;
    _lastFocalPoint = focalPoint;
    return true;
  }

  void applyScale({required double scale, required Offset focalPoint}) {
    _pinching = true;
    final worldBefore = worldPointAt(focalPoint);
    _zoom = scale.clamp(minZoom, maxZoom);
    _pan = focalPoint - (worldBefore * _zoom);
    _lastFocalPoint = focalPoint;
  }

  Offset worldPointAt(Offset screenPoint) => (screenPoint - _pan) / _zoom;

  void clampPan({required Size viewport, required Size world}) {
    final visible = Size(viewport.width / _zoom, viewport.height / _zoom);
    final maxX = ((world.width - visible.width) / 2).clamp(0, double.infinity);
    final maxY = ((world.height - visible.height) / 2).clamp(0, double.infinity);
    _pan = Offset(_pan.dx.clamp(-maxX, maxX), _pan.dy.clamp(-maxY, maxY));
  }

  void reset() {
    _zoom = 1;
    _pan = Offset.zero;
    _dragging = false;
    _pinching = false;
  }
}
```

- [ ] **Step 3: Integrate scale/pan and move world interaction to tap-up**

In `DefensePrototypeGame`, add a `BattlefieldCameraTransform`, publish its snapshot after each transform, use `ScaleUpdateInfo.pointerCount` to distinguish pinch from one-finger pan, and replace `onTapDown` with `onTapUp`:

```dart
final BattlefieldCameraTransform _cameraTransform = BattlefieldCameraTransform();

@override
void onScaleStart(ScaleStartInfo info) {
  _scaleStart = _cameraTransform.zoom;
  _cameraTransform.beginGesture(info.eventPosition.widget.toOffset());
}

@override
void onScaleUpdate(ScaleUpdateInfo info) {
  final focal = info.eventPosition.widget.toOffset();
  if (info.pointerCount >= 2) {
    _cameraTransform.applyScale(
      scale: _scaleStart * info.scale.global.x,
      focalPoint: focal,
    );
  } else {
    _cameraTransform.updatePan(focal);
  }
  _applyCameraTransform();
}

@override
void onTapUp(TapUpEvent event) {
  super.onTapUp(event);
  if (!_cameraTransform.suppressTap) {
    _handleTap(event.localPosition);
  }
}

void resetCamera() {
  _cameraTransform.reset();
  _applyCameraTransform();
}
```

`_applyCameraTransform()` updates `camera.viewfinder.zoom`, `camera.viewfinder.position`, clamps against `size`, and calls `sessionController.setCameraSnapshot(...)`. Remove the old immediate `onTapDown` placement path so dragging can never construct a tower or wall.
